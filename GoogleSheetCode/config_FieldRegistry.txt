/**
 * config_FieldRegistry.js
 *
 * سجلّ الحقول الواحد (Single Field Registry) — الحزمة 1.
 *
 * الغرض: مصدرٌ تصريحيّ واحد لكلّ خليّة ظاهرة في الأوراق الثلاث (الإعدادات،
 * القيود، التفضيلات). يشتقّ منه الباني صفوفه (هذه الحزمة)، وسيشتقّ منه القارئ
 * والمدقّق لاحقًا (الحزمتان 2+). الرابط الحقيقيّ هو `key` لا النصّ العربيّ —
 * فتغيّر العنوان لا يكسر شيئًا حين تتحوّل القراءة إلى المفتاح.
 *
 * ⚠️ في هذه الحزمة القرّاء (getSetting/readConstraints) ما زالوا يطابقون النصّ
 * العربيّ في العمود A؛ لذلك `label` هنا يُسحب حرفيًّا من CONSTANTS فيبني الباني
 * العناوين نفسها التي يبحث عنها القارئ — فيتطابقان بلا كسر.
 *
 * ⚠️ البذور (`seed`/`seedOnline`) مأخوذة من الورقة الحيّة (المقيسة من Drive)،
 * لا من ثوابت الكود؛ الحيّ هو الحقّ حيثما اختلفا. سابقًا كان يظهر سطرا دلتا
 * (SECOND_BREAK_AFTER الحيّ 5/الثابت 4، وLAST_HARD_MAX الحيّ 2/الثابت 3)؛
 * حُذف كلا الثابتين (الحزمتان 6/ج-2ب و6/ج-2ج) فصار تقرير الدلتا صفرًا.
 * دالّة rebuildAllSheetsFromRegistry تُبلِّغ عن كلّ اختلافٍ بين البذرة والثابت.
 *
 * قاعدة صلبة: حقلٌ status:'LIVE' بلا `consumer` = خطأ يوقف البناء
 * (_assertRegistryValid). لا يُعلَن حقلٌ حيًّا بلا مستهلكٍ مُثبَت.
 *
 * كلّ الدوالّ هنا تُقيَّم وقت الاستدعاء (لا في الأعلى) فيكون CONSTANTS متاحًا
 * مهما كان ترتيب تحميل ملفّات Apps Script.
 */

/**
 * نقطة الحقن الوحيدة لأوزان المحرّك (الحزمة 3): تُنتج خريطة الأوزان من قيم ورقة
 * القيود (model.constraints.params) — قيمةُ الورقة إن وُجدت، وإلّا **رميُ خطأ**
 * (لا تراجُع صامت للثابت؛ الحارس يمنع هذه الحالة أصلًا). يقرأ منها _softScore وwOf
 * وسائر المستهلكين حصرًا، فلا يبقى رقمٌ حرفيّ في مسار القرار.
 *
 * الحرّاس الصلبة الثلاثة وأوزان govt/dayBalance (لها خلايا في الإعدادات) تُدرَج من
 * الثوابت لاكتمال خريطة wOf فقط. الحرّاس مصدرُ حقيقتها الكود، والورقة مرآةٌ محميّة
 * يفرض المدقّق (_assertGuardsMatchCode) تطابقها. W_HOD_FIRST صار من الورقة (الحزمة 4).
 *
 * @param {Object} model ناتج loadScheduleData().
 * @return {Object} خريطة مفتاح الوزن → قيمته العدديّة.
 */
function getEngineWeights(model) {
  const P = (model && model.constraints && model.constraints.params) || {};
  const req = function (key) {
    if (!Object.prototype.hasOwnProperty.call(P, key)) {
      throw new Error('getEngineWeights: الوزن «' + key +
        '» غائبٌ عن ورقة القيود — الورقة أقدم من الكود. شغّل rebuildAllSheetsFromRegistry.');
    }
    const n = Number(P[key]);
    if (isNaN(n)) {
      throw new Error('getEngineWeights: الوزن «' + key + '» قيمته غير رقميّة: «' + P[key] + '».');
    }
    return n;
  };
  const BW = CONSTANTS.BREAK_WEIGHTS;
  const w = {};

  // من الورقة (الأقسام 4-7 LIVE)
  const sheetKeys = [
    'LAST_2ND_LIGHT_THRESHOLD', 'LAST_2ND_WEIGHT_LIGHT', 'LAST_2ND_WEIGHT_HEAVY',
    'W_SUPERVISOR', 'W_LAST_HARD', 'W_SUP_LAST', 'W_ADMIN_LAST', 'W_LAST_ALLOWED',
    'W_FIRST', 'W_HOD_FIRST', 'W_LAST_DESIRED', 'W_SPREAD', 'W_TRIPLE_FULL', 'W_TRIPLE_LIGHT',
    'W_TRIPLE_REPEAT', 'W_HOD_SIXTH_LOAD', 'W_TAIL_SUBJECT',
    'W_GAP_EACH', 'W_GAP_LARGE', 'W_GAP_HUGE',
    'W_DEPT_RESERVE', 'W_DEPT_HEAVY_BALANCE', 'BONUS_LOW_QUOTA_2ND_LAST', 'BONUS_LOW_QUOTA_THRESHOLD',
    'W_FIRST_LAST_SAME_DAY', 'W_SEVENTH_PRIORITY_FIRST',
    'W_ADJACENT_PAIR', 'W_PREFER_CONSEC_GAP',
    'W_ADJACENT_PAIR_RECESS', 'W_ADJACENT_PAIR_MEETING',
    'BROKEN_MATH_LAST', 'BROKEN_LAB', 'SUPERVISOR_DEFICIT', 'BROKEN_ADMIN_LAST',
    'BROKEN_SUP_LAST_OVER', 'BROKEN_NON_PRIORITY_2ND_LAST', 'BROKEN_DEPT_HEAD_NOT_FIRST',
    'BROKEN_DAILY_CAP', 'BROKEN_TEACHER_RULE', 'BROKEN_DAILY_FLOOR', 'BROKEN_DEPT_RESERVE',
    'TIME_BUDGET_MS', 'MAX_LOCAL_BACKTRACK_ATTEMPTS',
    'REPAIR_MAX_MOVES', 'REPAIR_MAX_SIDEWAYS', 'REPAIR_TABU_LEN',
    'RESERVE_PROBE_MAX_BUSY', 'RESERVE_PROBE_MAX_DEST',
    // العتبات البنيويّة (الحزمة 5) — عتبات «متى» لا أوزان «كم»
    'SUPERVISOR_MIN_FREE', 'BONUS_LOW_QUOTA_MIN_LOAD', 'HOD_SIXTH_MIN_LOAD',
    'GAP_LARGE_THRESHOLD', 'GAP_HUGE_THRESHOLD',
    'DEPT_RESERVE_LARGE_SIZE', 'DEPT_RESERVE_LARGE_MIN', 'DEPT_RESERVE_MID_SIZE', 'DEPT_RESERVE_MID_MIN',
    'HEAVY_DAY_MIN_LESSONS', 'HEAVY_DAY_CAP_DIVISOR',
  ];
  for (let i = 0; i < sheetKeys.length; i++) w[sheetKeys[i]] = req(sheetKeys[i]);

  // W_LAST_SAME_CLASS يُقرأ شرطيًّا: فقط حين «تفضيل صفّين مختلفين للأخيرتين» = «نعم»
  // (البناء المطفَأ الأثر — ما دام الزرّ «لا» فلا يُقرأ صفُّه أصلًا). req يرمي على الفراغ حين يُشغَّل.
  const _preferDiffLast = String(P.LAST_PREFER_DIFFERENT != null ? P.LAST_PREFER_DIFFERENT : '')
    .trim() === CONSTANTS.YES;
  w.W_LAST_SAME_CLASS = _preferDiffLast ? req('W_LAST_SAME_CLASS') : 0;

  // حرّاس: مصدر الحقيقة الكود، والورقة مرآةٌ محميّة يفرض المدقّق تطابقها.
  w.BROKEN_DEPT_HEAD_LAST     = BW.BROKEN_DEPT_HEAD_LAST;
  w.BROKEN_CONSEC_HARD        = BW.BROKEN_CONSEC_HARD;
  w.BROKEN_LAST_OVER_HARD     = BW.BROKEN_LAST_OVER_HARD;

  // govt/dayBalance (الحزمة 6/أ): تُشتقّ من نفس مصدر المستهلك الحقيقيّ
  // (model.settings عبر getGovtDaySettings/getDayBalanceSettings) لا من ثابتٍ
  // مُرمَّز — فلا نسخةٌ ثانية كاذبة في الخريطة. القيمة للعرض في تفكيك الكسور فقط
  // (العقوبة الفعليّة من ctx.govtLightDay/ctx.dayBalance مباشرةً).
  const gd  = (model && model.settings && model.settings.govtDay)    || {};
  const dbs = (model && model.settings && model.settings.dayBalance) || {};
  w.BROKEN_GOVT_LIGHTDAY_HARD = gd.wHard;
  w.BROKEN_GOVT_LIGHTDAY_SOFT = gd.wSoft;
  w.BROKEN_DAY_BALANCE_HARD   = dbs.wHard;
  w.BROKEN_DAY_BALANCE_SOFT   = dbs.wSoft;

  // «الأخيرتان في صفٍّ واحد» (نظير ما سبق): اسم الكسر يُشتقّ من نفس مصدر المستهلك
  // (W_LAST_SAME_CLASS أعلاه) لا من رقمٍ ثانٍ — للعرض في تفكيك الكسور فقط
  // (العقوبة الفعليّة من _lastSameClassPenalty مباشرةً). مطفأ ⟹ 0 كالوزن نفسه.
  w.BROKEN_LAST_SAME_CLASS_SOFT = w.W_LAST_SAME_CLASS;
  // شكل يوم المعلّم (الجلسة 59): اسما الكسرين مشتقّان من وزنَيهما — للعرض في تفكيك
  // الكسور فقط، لا رقمَ ثانٍ يُصان.
  // ⚠️ بعد تدرُّج السعر (الجلسة 61) صار هذا **سقفَ** الزوج لا سعرَه الفعليّ: الزوج
  // الذي تتخلّله فرصة أو أحد طرفيه اجتماع أرخص. المساهمة الحقيقيّة تقرؤها التقارير
  // من `_computeChangeableBreaks().shapePairs.penalty` لا من العدد × هذا الرقم.
  w.BROKEN_ADJACENT_PAIR = w.W_ADJACENT_PAIR;
  w.BROKEN_SINGLE_GAP    = w.W_PREFER_CONSEC_GAP;

  // «أولى وأخيرة في يومٍ واحد» (الجلسة 57): اسم الكسر يُشتقّ من الوزن نفسه — للعرض
  // في تفكيك الكسور فقط (العقوبة الفعليّة من _firstLastSameDayPenalty). 0 ⟹ مطفأ.
  w.BROKEN_FIRST_LAST_SAME_DAY = w.W_FIRST_LAST_SAME_DAY;

  return w;
}

/**
 * بذور جدول أقسام القيود (16 صفًّا). الأسماء من getAllDepartmentNames (نفس مصدر
 * القارئ فيتطابق المسح)، والعلَمان من الورقة الحيّة = الثوابت (لا انحراف):
 * «ممنوع» = نعم للرياضيات وحدها؛ «أولويّة» = نعم للأقسام الستّة.
 * @return {{name:string, banned:string, priority:string}[]}
 */
function getConstraintDeptSeeds() {
  const names = getAllDepartmentNames();
  const banned = CONSTANTS.LAST_PERIOD_RULES.BANNED_DEPARTMENTS;
  const priority = CONSTANTS.LAST_PERIOD_RULES.PRIORITY_DEPARTMENTS_FOR_SECOND;
  const out = [];
  for (let i = 0; i < names.length; i++) {
    out.push({
      name:     names[i],
      banned:   banned.indexOf(names[i]) >= 0 ? CONSTANTS.YES : CONSTANTS.NO,
      priority: priority.indexOf(names[i]) >= 0 ? CONSTANTS.YES : CONSTANTS.NO,
    });
  }
  return out;
}

/**
 * بذور جدول القواعد الفرديّة (24 صفًّا) — قتلًا لخطر المحو: كانت تُبنى فارغة فيمحوها
 * rebuildConstraints. القيم من الورقة الحيّة. الأعمدة بترتيب A..I:
 * targetType · target · ruleType · pattern · value · enforcement · weight · enabled · note.
 *
 * ⚠️ عمود «الوزن» (G) لصفوف المعلّمين السبعة يُبذَر فارغًا لا 0.5 — احترامًا لواجهة
 * الصدق التي يفرضها الباني (تحقّق «الوزن يجب أن يكون فارغًا لعدد الحصص الأخيرة»،
 * sheets_SheetBuilder:2577)؛ والقيمة محايدة سلوكيًّا (لا يقرؤها المحرّك للسقف
 * الفرديّ — يُنفَّذ من العمود «القيمة» E وحده). ART/الحاسوب يحتفظان بوزنهما (0.5/100).

  const PAT = CONSTANTS.CONSTRAINTS_SHEET.PATTERN_VOCAB;
  const ENF = CONSTANTS.CONSTRAINTS_SHEET.ENFORCEMENT_LEVELS;
  const TT  = CONSTANTS.CONSTRAINTS_SHEET.TARGET_TYPE_VOCAB;
  const YES = CONSTANTS.YES;
  const S = CONSTANTS.SUBJECTS;

  const RT_DAILY   = RTV[0];   // 'توزيع المادّة في اليوم'
  const RT_LASTCAP = RTV[5];   // 'عدد الحصص الأخيرة'
  const RT_DEPT_MEET = RTV[6]; // 'عدد حصص اجتماع القسم'
  const CONSEC     = PAT[0];   // 'متتاليتان'
  const HARD       = ENF[0];   // 'صلب'
  const SOFT       = ENF[1];   // 'مرن موزون'
  const SUBJ       = TT[1];    // 'مادّة'
  const TEACHER    = TT[0];    // 'معلّم'

  const seeds = [
    // كلتاهما «صلب» في الورقة الحيّة، والصلب لا يقرأ عمود الوزن (الوزن ميكانيزميّ
    // من BREAK_WEIGHTS.BROKEN_CONSEC_HARD) — فالوزنان هنا مرآةٌ للخليّة لا سلوك.
    // ⚠️ القيمة 2 (الجلسة 58): كتبها المالك في الورقة إسكاتًا لخطأ فحص التعارضات
    // («القيمة يجب أن تكون عددًا صحيحًا»). لا أثر لها في المحرّك — خريطة السقف
    // اليوميّ ترفع كلّ مادّة نمطُها «متتاليتان» إلى max(المشتقّ، 2) أصلًا
    // (algo_RepairEngine.js:221) — والبذرة تتبع الورقة لا الأثر.
    { targetType:SUBJ, target:S.ART.name,      ruleType:RT_DAILY,   pattern:CONSEC, value:2, enforcement:HARD, weight:500, enabled:YES, note:'' },
    { targetType:SUBJ, target:S.COMPUTER.name, ruleType:RT_DAILY,   pattern:CONSEC, value:2, enforcement:HARD, weight:100, enabled:YES, note:'' },
  ];
  // ⚠️ enabled = 'لا' (لا 'نعم'): السبعة خامدةٌ في الورقة الحيّة. إيقاظها يرفع سقف
  // حصصهم الأخيرة إلى 3 عبر lastCapByTeacher فيتغيّر الجدول — فالبذرة تتبع الورقة.
  const teacherNames = ['المقداد الزنكى','عباس الصفار','احمد احمد','محمد عاشور',
                        'حيدر الهزيم'];
  for (let i = 0; i < teacherNames.length; i++) {
    seeds.push({ targetType:TEACHER, target:teacherNames[i], ruleType:RT_LASTCAP,
      pattern:'', value:3, enforcement:SOFT, weight:'', enabled:CONSTANTS.NO, note:'' });
  }

  // ───────── الجلسة 62: مواءمة البذور للورقة الحيّة (فحص «⚖ مقارنة القيود») ─────────
  // كان في السبعة أعلاه «عماد رضوان» و«السعيد يوسف» ببذرة (3، مرن موزون، لا)،
  // وقد بدّلهما المالك في الورقة إلى سقفٍ **فعّال = 1**؛ فنُقلا إلى المجموعة
  // الحيّة أدناه لتتبع البذرةُ الورقةَ كما هي السياسة. وشهادةُ ذلك مستقلّة عن
  // الشاشة: سجلّ «فحص الجدوى» ١١ سبتمبر عدّ أصحاب السقف الفرديّ = 1 فذكرهما
  // ومعهما فاضل خورشيد وجمال حرش (ومجدى مدكور المبذور سلفًا).
  //
  // الشكل مطابقٌ لبذرة «مجدى مدكور» (الجلسة 58): العمودان F وG لا يُقرآن لهذا
  // النوع وهما فارغان في الورقة، فلا يُخترَع لهما شيء.
  const activeLastCap1 = ['عماد رضوان','السعيد يوسف','فاضل خورشيد','جمال حرش'];
  for (let i = 0; i < activeLastCap1.length; i++) {
    seeds.push({ targetType:TEACHER, target:activeLastCap1[i], ruleType:RT_LASTCAP,
      pattern:'', value:1, enforcement:'', weight:'', enabled:YES, note:'' });
  }

  // قيدا الجلسة 59 الشخصيّان لحيدر الهزيم — أضافهما المالك في الورقة بلا بذور.
  // كلاهما ثنائيٌّ بلا قيمة، ومستوى إلزامهما «مرن موزون» في الورقة الحيّة.
  // (قُرئا من مخرَج «⚖ مقارنة القيود بالسجلّ» ١٢ سبتمبر لا من تخمين: أوّل بذرةٍ
  //  كتبتُها لهما حملت قيمة 1 ومستوًى فارغًا فأبانها الفحص، فصُحّحت.)
  seeds.push({ targetType:TEACHER, target:'حيدر الهزيم', ruleType:RTV[11],

  // إعفاء عبدالكريم الصفار من يوم المراجعات (الجلسة 59) — وله صفٌّ ثانٍ «لا
  // حصّة أولى» مبذورٌ في ذيل الدالّة، فالمعلّم الواحد بقاعدتَين لا تعارض بينهما.
  seeds.push({ targetType:TEACHER, target:'عبدالكريم الصفار', ruleType:RTV[12],

  // «لا حصّة أولى» لفهد الوثيقي — **صلب** في الورقة. ⚠️ والمحرّك لا يقرأ عمود
  // الإلزام لهذا النوع: يُنفَّذ بكسر BROKEN_TEACHER_RULE ووزنُه حقلٌ عامّ في

  // صفوف الورقة الحيّة التي لم تكن في البذور (مطابقةٌ لها حرفًا بحرف حتّى تصير

  // ⚠️ «ابو» بألفٍ بلا همزة عمدًا: العمود B قائمةٌ منسدلة من أسماء ورقة المعلمين

  for (let i = 0; i < disabledLastCap.length; i++) {

  //

  //    الحيّة). الاسم هنا هو الموضع **الوحيد** في المشروع كلّه الذي يذكر معلّمًا
  //    بعينه لهذه القواعد؛ المحرّك عامّ لا يعرف اسمًا ولا عددًا.
  //    والاسم الذي لا يطابق اسمًا قصيرًا **يوقف التوليد** ببوّابة الجدوى

  //
  // ⚠️ الصفّ الثالث (إعفاء الاجتماع) ليس زينة: نافذته ستّ خانات ونصابه ستّ

  const RT_FREE_DAYS  = RTV[8];    // 'أيام التفرّغ'
  const RT_PERIODS    = RTV[9];    // 'الحصص المسموحة'
  const RT_MEETING_EX = RTV[10];   // 'إعفاء من اجتماع القسم'
  // الاسم القصير كما هو في منسدلة العمود B (لا الاسم الأوّل وحده) — الاسم الذي
  // لا يطابق اسمًا قصيرًا يوقف التوليد ببوّابة TEACHER_BLOCK_UNRESOLVED.
  const BLOCKED_TEACHER = 'مأمون الخطيب';
  const attendanceSeeds = [
    { ruleType:RT_FREE_DAYS,  value:'الإثنين، الأربعاء', note:'مهامّ خارجيّة — مفرَّغ يومين' },

  // (1) سقف السابعات = 1 لمعلّمي ثلاث موادّ أخرى (صلب) — نظير الفيزياء/الكيمياء.
  const hardSubjectLastCap2 = [S.BIOLOGY.name, S.GEOLOGY.name, S.ENVIRONMENTAL_EDUCATION.name];

  // rebuildConstraints غير هادمة. ترتيبها ترتيبُ الورقة (26، 27، 28).
  //
  // (1) الموسيقية كتلةٌ مزدوجة كالفنية — صلب، والوزن 300 مرآةُ الخليّة لا سلوك

/**
 * قاعدة صلبة: كلّ حقل LIVE (أو GUARD) يجب أن يحمل مستهلكًا مُثبَتًا. يرمي استثناءً
 * يوقف البناء إن وُجد حقلٌ حيٌّ أو حارسٌ بلا consumer.
 */
function _assertRegistryValid() {
  const reg = getFieldRegistry();
  const bad = [];
  for (let i = 0; i < reg.length; i++) {
    const f = reg[i];
    if ((f.status === 'LIVE' || f.status === 'GUARD') &&
        (!f.consumer || String(f.consumer).trim() === '')) {
      bad.push(f.key);
    }
  }
  if (bad.length > 0) {
    throw new Error('config_FieldRegistry: حقول أُعلِنت LIVE بلا مستهلك مُثبَت: ' + bad.join('، '));
  }
}
